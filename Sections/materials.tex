\subsection*{Lasers and Light Sources}
\label{sec:lightsources}
\begin{itemize}
    \item \textbf{Pulsed LASER:} PicoQuant PDL Series
    \begin{itemize}\footnotesize
        \item \textbf{PicoQuant Laser Diode Head}
        \item \textbf{Wavelength:} 640 nm
        \item \textbf{Repetition Rate ($f_{rep}$):} Tunable frequencies (operated between 1 MHz and 80 MHz).
        \item \textbf{FWHM of pulse:} $< 90\text{ ps}$
    \end{itemize}
    
    \item \textbf{Continuous wave (CW) laser 1}
    \begin{itemize}\footnotesize
       \item \textbf{Thorlabs S1FC660 (Fiber-coupled)}
       \item \textbf{Wavelength:} 660 nm
       \item \textbf{Power:} Tunable from 0 mW to 19.8 mW
       \item \textbf{Application:} Used for the Poissonian Statistics Verification.
    \end{itemize}

    \item \textbf{Continuous wave (CW) laser 2}
    \begin{itemize}\footnotesize
        \item \textbf{Thorlabs CLD1015:} Diode Controller.
        \item \textbf{Thorlabs DBR785P:} Diode head.
        \item \textbf{Wavelength:} 785 nm
        \item \textbf{Power:} Tunable diode current (between 62.7 mA and 65.5 mA)
        \item \textbf{Application:} Used for the Pseudothermal Light Simulator.
    \end{itemize}

    \item \textbf{Pseudothermal light simulator}
    \begin{itemize}\footnotesize
        \item \textbf{Light Source:} CW laser 2
        \item \textbf{Rotating Stage:} Thorlabs K10CR1/M 
        \item \textbf{Motor Speed:} Up to $20^\circ /s$
        \item \textbf{Diffuser:} Glass diffuser %[Insert Part Number]
    \end{itemize}
\end{itemize}

    \item \textbf{2 Pixel SPAD Module}
    \begin{itemize}\footnotesize
        \item \textbf{Model:} \href{https://novoviz.com/two-pixel-correlator/}{NV02TPX}
        \item \textbf{Array format:} 2 x 1 pixels.
        \item \textbf{Pixel Pitch:} $28 \mu m$ % Datasheet indicates 1100 um
        \item \textbf{Dark count rate:} 10 cps

    \item \textbf{Linear SPAD Array Module}
    \begin{itemize}\footnotesize
        \item \textbf{Model:} \href{https://novoviz.com/small-line/}{Novoviz Small Line Sensor}

        \item \textbf{Pixel Pitch:} $125 \mu m$
        \item \textbf{Quenching:} passive with active recharge
    \end{itemize}
    
    \item \textbf{Illumination Diffuser}
    \begin{itemize}\footnotesize

    \end{itemize}
